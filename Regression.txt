from sklearn.datasets import fetch_california_housing
from sklearn.model_selection import train_test_split
from sklearn.tree import DecisionTreeRegressor as D
from sklearn.metrics import mean_squared_error as m, mean_absolute_error as ma, r2_score as r

X,y = fetch_california_housing(return_X_y=True)
Xtr,Xte,ytr,yte = train_test_split(X,y,0.3,random_state=42)

p = D().fit(Xtr,ytr).predict(Xte)

print("MSE:", m(yte,p), "MAE:", ma(yte,p), "R2:", r(yte,p))
for t,p in zip(yte[:5],p[:5]): print(t,p)
