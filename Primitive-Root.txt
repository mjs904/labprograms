import java.util.*;

public class PrimitiveRootSimple {
    public static void main(String[] args) {
        Scanner s = new Scanner(System.in);

        System.out.print("Enter number: ");
        int p = s.nextInt();

        for (int g = 2; g < p; g++) {

            int value = 1;
            HashSet<Integer> set = new HashSet<>();

            for (int i = 1; i < p; i++) {
                value = (value * g) % p;
                set.add(value);
            }

            if (set.size() == p - 1) {
                System.out.println("Primitive Root: " + g);
                break;
            }
        }
    }
}
