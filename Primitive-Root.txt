import java.util.*;

public class PrimitiveRoot {
    public static void main(String[] a){
        Scanner s=new Scanner(System.in);

        System.out.print("Enter Number ");
        int p=s.nextInt();

        for(int g=2;g<p;g++){
            boolean[] h=new boolean[p];
            int v=1,c=0;

            for(int i=1;i<p;i++){
                v=(v*g)%p;
                if(h[v]){ c=-1; break; }
                h[v]=true; c++;
            }

            if(c==p-1){ System.out.println("Root:"+g); break; }
        }
    }
}
