from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split as t
from sklearn.linear_model import LogisticRegression as L
from sklearn.metrics import accuracy_score as a, classification_report as c

X,y = load_iris(return_X_y=True)
X,y = X[y<2], y[y<2]

Xtr,Xte,ytr,yte = t(X,y,0.3,random_state=42)

p = L().fit(Xtr,ytr).predict(Xte)

print("Accuracy:",a(yte,p))
print("\nClassification Report:\n",c(yte,p))
