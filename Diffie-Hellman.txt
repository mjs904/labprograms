        function pw(b,e,m){
            let r=1;
            for(let i=0;i<e;i++){
                r=(r*b)%m;
            }
            return r;
        }
        
        function dh(){
            let p = +document.getElementById("p").value;
            let g = +document.getElementById("q").value;
            let a = +document.getElementById("a").value;
            let b = +document.getElementById("b").value;
            
            

            if(!p || !g || !a || !b){
                document.getElementById("r").innerHTML = "please enter all values";
                return;
            }
            let A=pw(g,a,p);
            let B=pw(g,b,p);
            let K1=pw(B,a,p);
