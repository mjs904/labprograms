import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression as L
from sklearn.metrics import mean_squared_error as m, mean_absolute_error as ma, r2_score as r

np.random.seed(42)
X = 2*np.random.rand(100,1); y = 4+3*X+np.random.randn(100,1)

Xtr,Xte,ytr,yte = train_test_split(X,y,0.3,random_state=42)

mdl = L().fit(Xtr,ytr); p = mdl.predict(Xte)

print("MSE:",m(yte,p),"MAE:",ma(yte,p),"R2:",r(yte,p))
print("b0:",round(mdl.intercept_[0],2),"b1:",round(mdl.coef_[0][0],2))
