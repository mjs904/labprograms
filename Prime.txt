import java.util.*;

public class Prime {
    public static void main(String[] a){
        Scanner s=new Scanner(System.in);

        System.out.print("Enter number: ");
        int n=s.nextInt(), f=0;

        for(int i=2;i<=n/2;i++)
            if(n%i==0){ f=1; break; }

        System.out.println(n>1 && f==0 ? "Prime" : "Not Prime");
    }
}
